    Const _INSTANCE As String = "."
    Const _USER As String = "sa"
    Const _PWD As String = ""
    Const _BACKUPFILE As String = "c:\NorthwindBackup.bkp"
    Const _DATABASE As String = "Northwind"

    Dim oSQLServer As New SQLDMO.SQLServer
    Dim WithEvents oBackup As New SQLDMO.Backup
    Dim WithEvents oRestore As New SQLDMO.Restore

    Private Sub oBackup_PercentComplete(ByVal Message As String, ByVal Percent As Integer) Handles oBackup.PercentComplete
        showStatus(Percent)
    End Sub

    Private Sub btnBackup_Click(ByVal sender As System.Object, ByVal e As System.EventArgs) Handles btnBackup.Click
        Try
            Me.lblMessage.Text = "Backup process started.."
            Me.lblMessage.Refresh()
            showStatus(0)
            disableItems()
            doBackup()
        Catch ex As Exception
            MessageBox.Show(ex.Message, "Error", MessageBoxButtons.OK, MessageBoxIcon.Stop)
        Finally
            showStatus(0)
            Me.lblMessage.Text = ""
            Me.lblMessage.Refresh()
            disableItems(False)
            oSQLServer.DisConnect()
            Me.lblMessage.Text = "Backup process completed.."
            Me.lblMessage.Refresh()
        End Try
    End Sub

    Private Sub doBackup()
        With oBackup
            '.Devices = "[NorthwindBackup]"
            .Files = _BACKUPFILE
            .Database = _DATABASE
            .BackupSetName = "MyNorthwindBkp"
            .BackupSetDescription = "Backup from VB.NET application"
            oSQLServer.Connect(_INSTANCE, _USER, _PWD)
            .SQLBackup(oSQLServer)
        End With
        MessageBox.Show("Backup Completed Sucessfully", "Message", MessageBoxButtons.OK, MessageBoxIcon.Information)
    End Sub

    Private Sub doRestore()
        With oRestore
            '.Devices = "[NorthwindBackup]"
            .Files = _BACKUPFILE
            .Database = _DATABASE
            .ReplaceDatabase = True
            oSQLServer.Connect(_INSTANCE, _USER, _PWD)
            .SQLRestore(oSQLServer)
        End With
        MessageBox.Show("Restore Completed Successfully", "Message", MessageBoxButtons.OK, MessageBoxIcon.Information)
    End Sub

    Private Sub showStatus(ByVal percent As Integer)
        Me.ProgressBar1.Value = percent
        Me.ProgressBar1.Refresh()
        Me.lblPercent.Text = percent & "%"
        Me.lblPercent.Refresh()

        'this statement is only for better simulation for small databases..please remove it
        System.Threading.Thread.Sleep(2000)
    End Sub

    Private Sub disableItems(Optional ByVal enable As Boolean = True)
        Me.btnBackup.Enabled = Not enable
        Me.btnRestore.Enabled = Not enable
        Me.Refresh()
    End Sub

    Private Sub btnRestore_Click(ByVal sender As System.Object, ByVal e As System.EventArgs) Handles btnRestore.Click
        Try
            Me.lblMessage.Text = "Restore process started.."
            Me.lblMessage.Refresh()
            showStatus(0)
            disableItems()
            doRestore()
        Catch ex As Exception
            MessageBox.Show(ex.Message, "Error", MessageBoxButtons.OK, MessageBoxIcon.Stop)
        Finally
            showStatus(0)
            Me.lblMessage.Text = ""
            Me.lblMessage.Refresh()
            disableItems(False)
            oSQLServer.DisConnect()
            Me.lblMessage.Text = "Restore process completed.."
            Me.lblMessage.Refresh()
        End Try
    End Sub

    Private Sub oRestore_PercentComplete(ByVal Message As String, ByVal Percent As Integer) Handles oRestore.PercentComplete
        showStatus(Percent)
    End Sub
